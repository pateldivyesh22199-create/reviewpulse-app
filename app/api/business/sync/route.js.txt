import { NextResponse } from "next/server";
import { auth } from "@clerk/nextjs/server";
import { createClient } from "@supabase/supabase-js";

const supabase = createClient(process.env.NEXT_PUBLIC_SUPABASE_URL, process.env.SUPABASE_SERVICE_ROLE_KEY);

export async function POST(req) {
  try {
    const { userId } = await auth();
    const { businessId, googleLink, autopilot } = await req.json();
    const apiKey = process.env.GEMINI_API_KEY;

    if (!googleLink) return NextResponse.json({ error: "No Link" });

    // ૧. Serper API થી રિવ્યુ ખેંચવા
    const serperRes = await fetch("https://google.serper.dev/maps", {
      method: "POST",
      headers: { "X-API-KEY": process.env.SERPER_API_KEY, "Content-Type": "application/json" },
      body: JSON.stringify({ q: googleLink })
    });
    const serperData = await serperRes.json();
    const reviews = serperData.places?.[0]?.reviews || [];

    // ૨. બિઝનેસની વિગતો લેવી
    const { data: biz } = await supabase.from("businesses").select("*").eq("id", businessId).single();

    let processedCount = 0;

    for (const rev of reviews) {
      const { data: existing } = await supabase.from("reviews").select("id").eq("review_text", rev.text).maybeSingle();
      if (existing) continue;

      let aiReply = "Pending AI Response";
      
      // ૩. જો Autopilot ON હોય, તો તરત જ AI જવાબ લખશે
      if (autopilot) {
        const prompt = `You are AI Manager for ${biz.name}. Review: "${rev.text}" (${rev.rating} stars). Tone: ${biz.ai_tone}. Instructions: ${biz.custom_instructions}. Write a short professional reply.`;
        const aiRes = await fetch("https://api.groq.com/openai/v1/chat/completions", {
          method: "POST",
          headers: { "Authorization": `Bearer ${apiKey}`, "Content-Type": "application/json" },
          body: JSON.stringify({ messages: [{ role: "user", content: prompt }], model: "qwen/qwen3.8-27b" })
        });
        const aiData = await aiRes.json();
        aiReply = aiData.choices[0].message.content;
      }

      // ૪. સેવ કરવું
      await supabase.from("reviews").insert({
        user_id: userId, business_id: businessId, reviewer_name: rev.user,
        rating: rev.rating, review_text: rev.text, ai_response_text: aiReply,
        sentiment: rev.rating >= 4 ? 'positive' : 'negative'
      });
      processedCount++;
    }

    return NextResponse.json({ success: true, count: processedCount });
  } catch (err) { return NextResponse.json({ error: err.message }, { status: 500 }); }
}
